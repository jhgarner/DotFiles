local function hopper(opts, action)
  return function()
    local hop = require("hop")
    local directions = require("hop.hint").HintDirection

    if opts.direction == "before" then
      opts.direction = directions.BEFORE_CURSOR
    elseif opts.direction == "after" then
      opts.direction = directions.AFTER_CURSOR
    end
    setmetatable(opts or {}, { __index = hop.opts })

    local regex = action(hop, opts, require("hop.jump_regex"))
    local target = require("hop.jump_target").jump_target_generator(regex)

    hop.hint_with_callback(target, opts, function(jt)
      if vim.api.nvim_get_mode().mode == "no" then
        if opts.linewise then
          vim.cmd("normal! V")
        else
          vim.cmd("normal! v")
        end
      end
      hop.move_cursor_to(jt, hop.opts)
    end)
  end
end
